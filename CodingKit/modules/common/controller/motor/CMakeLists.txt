cmake_minimum_required(VERSION 3.16)
project(motor_primitives C)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/ControllerPrimitives.cmake")
add_library(motor_primitives motor.c)
target_include_directories(motor_primitives PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(motor_primitives PUBLIC controller_primitives)
set_target_properties(motor_primitives PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(motor_primitives PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    find_program(MOTOR_HOST_GCC NAMES gcc REQUIRED)
    add_test(NAME motor_golden COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/run_tests.py"
        --cc "${MOTOR_HOST_GCC}" --output "${CMAKE_CURRENT_BINARY_DIR}/golden")
endif()
