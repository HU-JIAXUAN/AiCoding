foreach(test_name if_control if_rl)
    add_executable("${test_name}" "test_${test_name}.c")
    target_link_libraries("${test_name}" PRIVATE if_controller)
    set_target_properties("${test_name}" PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options("${test_name}" PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
    add_test(NAME "${test_name}" COMMAND "${test_name}")
endforeach()
