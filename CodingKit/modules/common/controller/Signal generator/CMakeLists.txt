cmake_minimum_required(VERSION 3.16)
project(signal_generator LANGUAGES C)

add_library(signal_generator STATIC signal_gen.c)
target_include_directories(signal_generator PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(signal_generator PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(NOT MSVC)
    target_link_libraries(signal_generator PUBLIC m)
endif()

option(SIGNAL_GEN_BUILD_TESTS "Build focused signal generator host checks and example" OFF)
if(SIGNAL_GEN_BUILD_TESTS)
    enable_testing()
    add_executable(signal_gen_test tests/test_signal_gen.c)
    add_executable(signal_gen_example examples/minimal_example.c)
    target_link_libraries(signal_gen_test PRIVATE signal_generator)
    target_link_libraries(signal_gen_example PRIVATE signal_generator)
    foreach(target signal_generator signal_gen_test signal_gen_example)
        set_target_properties(${target} PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
        if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
        endif()
    endforeach()
    add_test(NAME signal_gen_numeric COMMAND signal_gen_test)
endif()
