cmake_minimum_required(VERSION 3.10)
project(pid_controller C)
set(CMAKE_C_STANDARD 99)
add_library(pid_controller src/pid.c src/pid_pi.c)
target_include_directories(pid_controller PUBLIC src)
set_target_properties(pid_controller PROPERTIES C_STANDARD_REQUIRED YES)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(BUILD_TESTING)
    add_executable(pid_runtime tests/test_pid_runtime.c)
    target_link_libraries(pid_runtime PRIVATE pid_controller)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(pid_runtime PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
    add_test(NAME pid_runtime COMMAND pid_runtime)
endif()
